export default async function handler(req, res) { if (req.method !== 'POST') { return res.status(405).json({ error: '허용되지 않은 요청 방식입니다.' }); } const apiKey = process.env.FAL_KEY; if (!apiKey) { return res.status(500).json({ error: '서버에 API 키가 설정되지 않았습니다.' }); } try { const { image_url, prompt } = req.body; const response = await fetch("https://fal.run", { method: "POST", headers: { "Authorization": `Key ${apiKey}`, "Content-Type": "application/json", "X-Fal-Store-IO": "0", "X-Fal-Object-Lifecycle-Preference": JSON.stringify({ expiration_duration_seconds: 60 }) }, body: JSON.stringify({ image_url, prompt, guidance_scale: 3.5, num_inference_steps: 30, output_format: "jpeg" }) }); const data = await response.json(); if (!response.ok) { return res.status(response.status).json({ error: data.detail || 'AI 합성 처리 중 오류가 발생했습니다.' }); } return res.status(200).json(data); } catch (error) { return res.status(500).json({ error: '서버 내부 통신 오류가 발생했습니다.' }); } }
